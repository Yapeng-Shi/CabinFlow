cmake_minimum_required(VERSION 3.22)
project(CabinFlowVoiceDemo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
get_filename_component(CABINFLOW_REPO_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include(${CABINFLOW_REPO_DIR}/runtime/cmake/Sanitizers.cmake)
add_subdirectory(${CABINFLOW_REPO_DIR}/runtime ${CMAKE_CURRENT_BINARY_DIR}/runtime)
add_subdirectory(${CABINFLOW_REPO_DIR}/agent/inference ${CMAKE_CURRENT_BINARY_DIR}/inference)

add_subdirectory(${CABINFLOW_REPO_DIR}/agent/pipeline ${CMAKE_CURRENT_BINARY_DIR}/pipeline)
add_executable(voice_demo main.cpp)
target_link_libraries(voice_demo PRIVATE CabinFlow::VoicePipeline CabinFlow::NativeBackends)
cabinflow_enable_compiler_warnings(voice_demo)
add_executable(voice_demo_server server_main.cpp)
target_link_libraries(voice_demo_server PRIVATE CabinFlow::VoicePipeline CabinFlow::NativeBackends
    CabinFlow::Gateway CabinFlow::Runtime CabinFlow::Net)
cabinflow_enable_compiler_warnings(voice_demo_server)
add_executable(rag_probe rag_probe.cpp)
target_include_directories(rag_probe PRIVATE
    ${CABINFLOW_INFERENCE_DIR}/llama.cpp-a7a98e0fffed794396b3fbad4dcdbbc184963645/vendor)
target_link_libraries(rag_probe PRIVATE CabinFlow::NativeBackends)
cabinflow_enable_compiler_warnings(rag_probe)
add_subdirectory(frontend)
