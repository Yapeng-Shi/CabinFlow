cmake_minimum_required(VERSION 3.22)
project(CabinFlowVoiceDemo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
get_filename_component(CABINFLOW_REPO_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include(${CABINFLOW_REPO_DIR}/runtime/cmake/Sanitizers.cmake)
add_subdirectory(${CABINFLOW_REPO_DIR}/runtime ${CMAKE_CURRENT_BINARY_DIR}/runtime)
add_subdirectory(${CABINFLOW_REPO_DIR}/agent/inference ${CMAKE_CURRENT_BINARY_DIR}/inference)
add_subdirectory(${CABINFLOW_REPO_DIR}/agent/pipeline ${CMAKE_CURRENT_BINARY_DIR}/pipeline)
add_executable(voice_demo main.cpp)
target_link_libraries(voice_demo PRIVATE CabinFlow::VoicePipeline CabinFlow::NativeBackends)
cabinflow_enable_compiler_warnings(voice_demo)
if(BUILD_TESTING)
    add_executable(voice_pipeline_test ${CABINFLOW_REPO_DIR}/agent/tests/voice_pipeline_test.cpp)
    target_link_libraries(voice_pipeline_test PRIVATE CabinFlow::VoicePipeline)
    cabinflow_enable_compiler_warnings(voice_pipeline_test)
    add_test(NAME voice_pipeline_test COMMAND voice_pipeline_test)
    set_tests_properties(voice_pipeline_test PROPERTIES TIMEOUT 30)
    add_executable(wav_contract_test ${CABINFLOW_REPO_DIR}/agent/tests/wav_contract_test.cpp)
    target_include_directories(wav_contract_test PRIVATE ${CABINFLOW_REPO_DIR}/agent/inference/src)
    target_link_libraries(wav_contract_test PRIVATE CabinFlow::NativeBackends)
    cabinflow_enable_compiler_warnings(wav_contract_test)
    add_test(NAME wav_contract_test COMMAND wav_contract_test)
    set_tests_properties(wav_contract_test PROPERTIES TIMEOUT 10)
endif()
