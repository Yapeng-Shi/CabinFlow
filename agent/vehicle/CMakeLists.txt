cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(CabinFlowVehicle LANGUAGES CXX)
    include(CTest)
    include(${CMAKE_CURRENT_SOURCE_DIR}/../../runtime/cmake/CompilerWarnings.cmake)
endif()

find_package(Threads REQUIRED)
add_library(cabinflow_vehicle src/vehicle_backend.cpp src/vehicle_can_codec.cpp
    src/socket_can_vehicle.cpp src/vcan_interface.cpp)
add_library(CabinFlow::Vehicle ALIAS cabinflow_vehicle)
target_include_directories(cabinflow_vehicle PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(cabinflow_vehicle PUBLIC cxx_std_17)
target_link_libraries(cabinflow_vehicle PUBLIC Threads::Threads)
cabinflow_enable_compiler_warnings(cabinflow_vehicle)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../apps/vehicle_ecu ${CMAKE_CURRENT_BINARY_DIR}/vehicle-ecu)

if(BUILD_TESTING)
    add_library(cabinflow_test_vehicle src/fake_vehicle.cpp)
    add_library(CabinFlow::TestVehicle ALIAS cabinflow_test_vehicle)
    target_link_libraries(cabinflow_test_vehicle PUBLIC CabinFlow::Vehicle)
    cabinflow_enable_compiler_warnings(cabinflow_test_vehicle)
    add_executable(vehicle_can_test ${CMAKE_CURRENT_SOURCE_DIR}/../tests/vehicle_can_test.cpp)
    target_link_libraries(vehicle_can_test PRIVATE CabinFlow::Vehicle CabinFlow::TestVehicle)
    cabinflow_enable_compiler_warnings(vehicle_can_test)
    add_test(NAME vehicle_can_unit_test COMMAND vehicle_can_test --unit)
    set_tests_properties(vehicle_can_unit_test PROPERTIES TIMEOUT 10)
endif()
